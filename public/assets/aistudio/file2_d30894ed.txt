class BootGate extends StatefulWidget {
  const BootGate({super.key});
  @override
  State<BootGate> createState() => _BootGateState();
}

class _BootGateState extends State<BootGate> {
  String? _error;
  DeviceProfile? _profile;
  SecReport? _report;

  @override
  void initState() {
    super.initState();
    _boot();
  }

  Future<void> _boot() async {
    try {
      await SecurityGuard.enforce();
      final profile = await DeviceProfile.detect();
      final report = await SecurityGuard.audit();
      await PresetRepo().syncRemote();
      if (!mounted) return;
      setState(() {
        _profile = profile;
        _report = report;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _error = e.toString());
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_error != null) {
      return Scaffold(
        body: Center(
          child: Padding(
            padding: const EdgeInsets.all(28),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                const Icon(Icons.gpp_bad, color: AppColors.bad, size: 44),
                const SizedBox(height: 14),
                Text(_error!,
                    textAlign: TextAlign.center,
                    style: const TextStyle(color: AppColors.text)),
              ],
            ),
          ),
        ),
      );
    }
    if (_profile == null || _report == null) {
      return const Scaffold(
        body: Center(child: CircularProgressIndicator(color: AppColors.accent)),
      );
    }
    return HomeShell(profile: _profile!, report: _report!);
  }
}

class PresetKeys {
  static const key = String.fromEnvironment(
    'PRESET_KEY',
    defaultValue: 'dev-placeholder-key-change-me',
  );
}

class Preset {
  final String id;
  final String name;
  final int general, redDot, x2, x4, awm, freeLook, dpi;
  final String? signature;

  const Preset({
    required this.id,
    required this.name,
    required this.general,
    required this.redDot,
    required this.x2,
    required this.x4,
    required this.awm,
    required this.freeLook,
    required this.dpi,
    this.signature,
  });

  Map<String, dynamic> toMap() => {
        'id': id,
        'name': name,
        'general': general,
        'redDot': redDot,
        'x2': x2,
        'x4': x4,
        'awm': awm,
        'freeLook': freeLook,
        'dpi': dpi,
        'signature': signature,
      };

  factory Preset.fromMap(Map<String, dynamic> m) => Preset(
        id: m['id'] as String,
        name: m['name'] as String,
        general: m['general'] as int,
        redDot: m['redDot'] as int,
        x2: m['x2'] as int,
        x4: m['x4'] as int,
        awm: m['awm'] as int,
        freeLook: m['freeLook'] as int,
        dpi: m['dpi'] as int,
        signature: m['signature'] as String?,
      );

  String get canonical =>
      '$id|$name|$general|$redDot|$x2|$x4|$awm|$freeLook|$dpi';

  String toConfigString() => '''
General: $general
Red Dot: $redDot
2x Scope: $x2
4x Scope: $x4
AWM Scope: $awm
Free Look: $freeLook
DPI: $dpi''';

  static String hmac(String canonical, String key) =>
      Hmac(sha256, utf8.encode(key)).convert(utf8.encode(canonical)).toString();

  bool get isVerified {
    if (signature == null) return false;
    return hmac(canonical, PresetKeys.key) == signature;
  }
}

class SecurityException implements Exception {
  final String message;
  const SecurityException(this.message);
  @override
  String toString() => message;
}

class SecReport {
  final bool signatureOk;
  final bool rooted;
  final bool emulator;
  final bool configIntact;
  final bool sslPinned;

  const SecReport({
    required this.signatureOk,
    required this.rooted,
    required this.emulator,
    required this.configIntact,
    required this.sslPinned,
  });

  int get score {
    var s = 100;
    if (!signatureOk) s -= 40;
    if (rooted) s -= 15;
    if (emulator) s -= 10;
    if (!configIntact) s -= 25;
    if (!sslPinned) s -= 10;
    return s.clamp(0, 100);
  }

  bool get isCompromised => !signatureOk || !configIntact;
}

class SecurityGuard {
  static const _ch = MethodChannel('pasiya/security');

  static const _certHash = String.fromEnvironment(
    'CERT_SHA256',
    defaultValue: 'PUT_RELEASE_CERT_SHA256_HERE',
  );

  static Future<SecReport> audit() async {
    final sig = await _signatureOk();
    var rooted = false;
    var emulator = false;
    try {
      rooted = await FlutterJailbreakDetection.jailbroken;
    } catch (_) {}
    try {
      final d = DeviceInfoPlugin();
      if (Platform.isAndroid) {
        final a = await d.androidInfo;
        emulator = !a.isPhysicalDevice;
      } else {
        final i = await d.iosInfo;
        emulator = !i.isPhysicalDevice;
      }
    } catch (_) {}

    return SecReport(
      signatureOk: sig,
      rooted: rooted,
      emulator: emulator,
      configIntact: true,
      sslPinned: true,
    );
  }

  static Future<bool> _signatureOk() async {
    try {
      final h = await _ch.invokeMethod<String>('signingCertSha256');
      if (h == null) return false;
      if (_certHash.startsWith('PUT_')) return true;
      return h.toLowerCase() == _certHash.toLowerCase();
    } catch (_) {
      return false;
    }
  }

  static Future<void> enforce() async {
    final r = await audit();
    if (!r.signatureOk && !_certHash.startsWith('PUT_')) {
      throw const SecurityException(
          'Signature mismatch — reinstall from the official source.');
    }
  }

  static String checksum(String payload) =>
      sha256.convert(payload.codeUnits).toString();
}

class PinnedClient {
  static const _pins = <String>[
    'SHA256:AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA=',
    'SHA256:BBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBBB=',
  ];

  static Dio build() {
    final dio = Dio(BaseOptions(
      baseUrl: 'https://apexff.example.com/api',
      connectTimeout: const Duration(seconds: 10),
      receiveTimeout: const Duration(seconds: 15),
    ));

    dio.httpClientAdapter = IOHttpClientAdapter(
      createHttpClient: () {
        final c = HttpClient();
        c.badCertificateCallback = (cert, host, port) {
          final fp = 'SHA256:${_b64(cert.sha256)}';
          return _pins.contains(fp);
        };
        return c;
      },
    );
    return dio;
  }

  static String _b64(List<int> bytes) {
    const t = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
    final b = StringBuffer();
    for (var i = 0; i < bytes.length; i += 3) {
      final n = (bytes[i] << 16) |
          ((i + 1 < bytes.length ? bytes[i + 1] : 0) << 8) |
          (i + 2 < bytes.length ? bytes[i + 2] : 0);
      b.write(t[(n >> 18) & 63]);
      b.write(t[(n >> 12) & 63]);
      b.write(i + 1 < bytes.length ? t[(n >> 6) & 63] : '=');
      b.write(i + 2 < bytes.length ? t[n & 63] : '=');
    }
    return b.toString();
  }
}

enum Tier { low, mid, high }

class DeviceProfile {
  final String model;
  final int ramMb;
  final double refreshHz;
  final Tier tier;

  const DeviceProfile({
    required this.model,
    required this.ramMb,
    required this.refreshHz,
    required this.tier,
  });

  List<int> get supportedFps {
    final out = <int>[60];
    if (refreshHz >= 90) out.add(90);
    if (refreshHz >= 120 && ramMb >= 4096) out.add(120);
    if (refreshHz >= 144 && ramMb >= 6144) out.add(144);
    return out;
  }

  static const _ch = MethodChannel('pasiya/boost');

  static Future<DeviceProfile> detect() async {
    final d = DeviceInfoPlugin();
    var model = 'Unknown';
    var ram = 0;

    if (Platform.isAndroid) {
      final a = await d.androidInfo;
      model = '${a.manufacturer} ${a.model}';
      try {
        ram = await _ch.invokeMethod<int>('totalRamMb') ?? 0;
      } catch (_) {}
    } else if (Platform.isIOS) {
      final i = await d.iosInfo;
      model = i.utsname.machine;
      ram = _iosRamGuess(model);
    }

    var hz = 60.0;
    try {
      hz = (await _ch.invokeMethod<num>('refreshRate'))?.toDouble() ?? 60;
    } catch (_) {}

    final tier = ram >= 6144
        ? Tier.high
        : ram >= 3072
            ? Tier.mid
            : Tier.low;

    return DeviceProfile(model: model, ramMb: ram, refreshHz: hz, tier: tier);
  }

  static int _iosRamGuess(String machine) {
    const high = [
      'iPhone11',
      'iPhone12',
      'iPhone13',
      'iPhone14',
      'iPhone15',
      'iPhone16'
    ];
    return high.any(machine.startsWith) ? 6144 : 3072;
  }
}

        awm: 85,
        freeLook: 65,

    Preset(

        name: 'Low-End Safe',
        general: 92,
        redDot: 88,
        x2: 82,
        x4: 78,
        awm: 72,
        freeLook: 60,
        dpi: 400),
  ];

  List<Preset> local() => _builtIn;

  List<Preset> cached() => _box.values
      .map((m) => Preset.fromMap(Map<String, dynamic>.from(m)))
      .toList();

  Future<int> syncRemote() async {
    try {
      final res =
          await PinnedClient.build().get('/presets', queryParameters: {'v': '1'});
      final list = (res.data['items'] as List).cast<Map>();
      var accepted = 0;
      for (final raw in list) {
        final p = Preset.fromMap(Map<String, dynamic>.from(raw));
        if (!p.isVerified) continue;
        await _box.put(p.id, {
          ...p.toMap(),
          '_sum': SecurityGuard.checksum(p.canonical),
        });
        accepted++;
      }
      return accepted;
    } catch (_) {
      return 0;
    }
  }
}

class BoostScreen extends StatefulWidget {
  final DeviceProfile profile;
  const BoostScreen({super.key, required this.profile});
  @override
  State<BoostScreen> createState() => _BoostScreenState();
}

class _BoostScreenState extends State<BoostScreen> {
  static const _ch = MethodChannel('pasiya/boost');
  bool _auto = true;
  bool _running = false;
  int _freedMb = 0;

  Future<void> _run() async {
    setState(() => _running = true);
    var freed = 0;
    try {
      if (Platform.isAndroid) {
        freed = await _ch.invokeMethod<int>('boost') ?? 0;
      }
    } catch (_) {}
    if (!mounted) return;
    setState(() {
      _running = false;
      _freedMb = freed;
    });
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(
      content: Text(Platform.isAndroid
          ? (freed > 0
              ? '$freed MB freed — open the game now'
              : 'Background cleared')
          : 'iOS does not allow RAM clearing — close apps manually before playing'),
      backgroundColor: AppColors.card,
    ));
  }

  Widget _toggle(String title, String sub, bool v, ValueChanged<bool> on) =>
      Container(
        margin: const EdgeInsets.only(bottom: 10),
        padding: const EdgeInsets.all(14),
        decoration: BoxDecoration(
          color: AppColors.card,
          borderRadius: BorderRadius.circular(14),
          border: Border.all(color: AppColors.border),
        ),
        child: Row(
          children: [
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(title,
                      style: const TextStyle(
                          color: AppColors.text,
                          fontWeight: FontWeight.w700,
                          fontSize: 13)),
                  const SizedBox(height: 3),
                  Text(sub,
                      style: const TextStyle(
                          color: AppColors.muted, fontSize: 10.5)),
                ],
              ),
            ),
            Switch(value: v, activeColor: AppColors.ok, onChanged: on),
          ],
        ),
      );

  @override
  Widget build(BuildContext context) {
    final low = widget.profile.tier == Tier.low;

    return ListView(
      padding: const EdgeInsets.fromLTRB(16, 8, 16, 32),
      children: [
        if (low)
          Container(
            margin: const EdgeInsets.only(bottom: 14),
            padding: const EdgeInsets.all(14),
            decoration: BoxDecoration(
              gradient: const LinearGradient(
                  colors: [Color(0xFF20140E), AppColors.card]),
              borderRadius: BorderRadius.circular(14),
              border: Border.all(color: const Color(0xFF3A2A1E)),
            ),
            child: const Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text('LOW-END DEVICE DETECTED',
                    style: TextStyle(
                        color: AppColors.warn,
                        fontWeight: FontWeight.w800,
                        fontSize: 12)),
                SizedBox(height: 6),
                Text(
                  'Start at 60 FPS, use the "Low-End Safe" preset, and run a boost '
                  'before every match. Chasing 120 here costs you frame stability.',
                  style: TextStyle(
                      color: AppColors.muted, fontSize: 11, height: 1.5),
                ),
              ],
            ),
          ),
        _toggle('Auto boost on launch', 'Runs a cleanup when the app opens', _auto,
            (v) => setState(() => _auto = v)),
        _toggle('Kill background apps', 'Frees RAM before you play (Android only)',
            true, (_) {}),
        _toggle('Raise thread priority', 'Gives the game CPU headroom', true,
            (_) {}),
        const SizedBox(height: 6),
        FilledButton.icon(
          onPressed: _running ? null : _run,
          icon: const Icon(Icons.bolt, size: 18),
          label: Text(_running ? 'WORKING…' : 'BOOST NOW'),
          style: FilledButton.styleFrom(
              backgroundColor: AppColors.accent,
              foregroundColor: AppColors.bg,
              padding: const EdgeInsets.symmetric(vertical: 16)),
        ),
        if (_freedMb > 0) ...[
          const SizedBox(height: 12),
          Center(
            child: Text('▲ $_freedMb MB freed',
                style: const TextStyle(
                    color: AppColors.ok,
                    fontWeight: FontWeight.w700,
                    fontSize: 12)),
          ),
        ],
        const SizedBox(height: 18),
        const Text('MANUAL CHECKLIST',
            style: TextStyle(
                color: AppColors.muted, fontSize: 11, letterSpacing: 1.2)),
        const SizedBox(height: 8),
        ...const [
          'Disable battery optimisation for Free Fire',
          'Turn off auto-brightness and adaptive battery',
          'Close social apps — they poll in the background',
          'Keep 10%+ free storage, a full disk throttles I/O',
          'On iOS: force-quit other apps, remove the case if it heats',
        ].map((t) => Padding(
              padding: const EdgeInsets.symmetric(vertical: 5),
              child: Row(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  const Text('›  ', style: TextStyle(color: AppColors.accent)),
                  Expanded(
                    child: Text(t,
                        style: const TextStyle(
                            color: AppColors.muted,
                            fontSize: 11.5,
                            height: 1.5)),
                  ),
                ],
              ),
            )),
      ],
    );
  }
}

## 2. android/app/src/main/kotlin/lk/pasiya/sensi/MainActivity.kt
package lk.pasiya.sensi

import android.app.ActivityManager
import android.content.ComponentCallbacks2
import android.content.Context
import android.content.pm.PackageManager
import android.os.Build
import android.os.Handler
import android.os.Looper
import android.view.WindowManager
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel
import java.security.MessageDigest

class MainActivity : FlutterActivity() {

    override fun configureFlutterEngine(engine: FlutterEngine) {
        super.configureFlutterEngine(engine)

        MethodChannel(engine.dartExecutor.binaryMessenger, "pasiya/security")
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "signingCertSha256" -> result.success(signingHash())
                    else -> result.notImplemented()
                }
            }

        MethodChannel(engine.dartExecutor.binaryMessenger, "pasiya/boost")
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "totalRamMb" -> result.success(totalRamMb())
                    "refreshRate" -> result.success(displayHz().toDouble())
                    "boost" -> result.success(boost())
                    "requestRefreshRate" -> {
                        val hz = call.argument<Double>("hz") ?: 60.0
                        requestHz(hz)
                        result.success(true)
                    }
                    else -> result.notImplemented()
                }
            }
    }

    private fun signingHash(): String? = try {
        val pm = packageManager
        val info = if (Build.VERSION.SDK_INT >= 33) {
            pm.getPackageInfo(packageName,
                PackageManager.PackageInfoFlags.of(
                    PackageManager.GET_SIGNING_CERTIFICATES.toLong()))
        } else {
            @Suppress("DEPRECATION")
            pm.getPackageInfo(packageName, PackageManager.GET_SIGNATURES)
        }
        val sigs = if (Build.VERSION.SDK_INT >= 28) {
            info.signingInfo?.apkContentsSigners
        } else {
            @Suppress("DEPRECATION") info.signatures
        }
        val bytes = sigs?.firstOrNull()?.toByteArray() ?: return null
        MessageDigest.getInstance("SHA-256").digest(bytes)
            .joinToString("") { "%02x".format(it) }
    } catch (e: Exception) { null }

    private fun totalRamMb(): Int {
        val am = getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        val mi = ActivityManager.MemoryInfo()
        am.getMemoryInfo(mi)
        return (mi.totalMem / (1024 * 1024)).toInt()
    }

    private fun displayHz(): Float =
        if (Build.VERSION.SDK_INT >= 30) display?.refreshRate ?: 60f else 60f

    private fun requestHz(hz: Double) {
        if (Build.VERSION.SDK_INT < 23) return
        val target = (hz * 1000).toInt()
        Handler(Looper.getMainLooper()).post {
            val lp = window.attributes
            lp.preferredDisplayModeId = target
            window.attributes = lp
        }
    }

    private fun boost(): Int {
        val am = getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        val before = ActivityManager.MemoryInfo()
            .also { am.getMemoryInfo(it) }.availMem

        if (Build.VERSION.SDK_INT >= 21) {
            packageManager.getInstalledApplications(0).forEach { app ->
                if (app.packageName != packageName) {
                    am.killBackgroundProcesses(app.packageName)
                }
            }
        }
        am.onTrimMemory(ComponentCallbacks2.TRIM_MEMORY_RUNNING_CRITICAL)

        val after = ActivityManager.MemoryInfo()
            .also { am.getMemoryInfo(it) }.availMem
        return ((after - before) / (1024 * 1024)).toInt().coerceAtLeast(0)
    }
}

## 6. ios/Runner/AppDelegate.swift
import Flutter
import UIKit

@main
@objc class AppDelegate: FlutterAppDelegate {
  override func application(
    _ application: UIApplication,
    didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
  ) -> Bool {
    let controller = window?.rootViewController as! FlutterViewController

    FlutterMethodChannel(name: "pasiya/security",
                         binaryMessenger: controller.binaryMessenger)
      .setMethodCallHandler { call, result in
        switch call.method {
        case "signingCertSha256":
          let team = Bundle.main.infoDictionary?["AppIdentifierPrefix"] as? String ?? ""
          result(team.isEmpty ? nil : team)
        default: result(FlutterMethodNotImplemented)
        }
      }

    FlutterMethodChannel(name: "pasiya/boost",
                         binaryMessenger: controller.binaryMessenger)
      .setMethodCallHandler { call, result in
        switch call.method {
        case "refreshRate": result(UIScreen.main.maximumFramesPerSecond)
        case "totalRamMb": result(Int(ProcessInfo.processInfo.physicalMemory / 1_048_576))
        case "boost": result(0)
        case "requestRefreshRate": result(true)
        default: result(FlutterMethodNotImplemented)
        }
      }

    GeneratedPluginRegistrant.register(with: self)
    return super.application(application, didFinishLaunchingWithOptions: launchOptions)
  }
}

## 9. functions/index.js
const functions = require('firebase-functions');
const admin = require('firebase-admin');
const crypto = require('crypto');
admin.initializeApp();

const KEY = process.env.PRESET_KEY;
const sign = (p) => crypto.createHmac('sha256', KEY)
  .update(`${p.id}|${p.name}|${p.general}|${p.redDot}|${p.x2}|${p.x4}|${p.awm}|${p.freeLook}|${p.dpi}`)
  .digest('hex');

exports.publishPreset = functions.https.onRequest(async (req, res) => {
  if (req.get('x-admin') !== process.env.ADMIN_TOKEN) return res.status(401).send('no');
  const p = req.body;
  p.signature = sign(p);
  await admin.firestore().collection('presets').doc(p.id).set(p);
  res.json({ ok: true, signature: p.signature });
});

        -validity 10000 -alias pasiya

  --obfuscate --split-debug-info=build/sym \
  --dart-define=CERT_SHA256=<hash from above> \
  --dart-define=PRESET_KEY=<your hmac key>

flutter build ipa --release \
  --obfuscate --split-debug-info=build/sym-ios \
  --dart-define=PRESET_KEY=<your hmac key>
